openrewrite converts source code to Lossless Semantic Trees.

For instance, a ClassDeclaration is an LST that defines a class. 
A typical class declaration will be composed of elements such as fields, 
methods, constructors, and inner classes. Each of those elements are, 
themselves, an LST. So the term "LST" may refer to an entire, 
complete Java file or just one piece of it.

public String getToken(String salesforceInstanceUrl, 
                       String clientId, 
                       String clientSecret, 
                       String userName, 
                       String password, 
                       String orgName) {

    // Prepare credentials
    MultiValueMap<String, String> credentials = new LinkedMultiValueMap<>();
    credentials.add("grant_type", "password");
    credentials.add("client_id", clientId);
    credentials.add("client_secret", clientSecret);
    credentials.add("username", userName);
    credentials.add("password", password);

    // Build OAuth URL
    URI oAuthURL = UriComponentsBuilder
            .fromHttpUrl(salesforceInstanceUrl + "/services/oauth2/token")
            .build()
            .encode()
            .toUri();

    // Create RestClient
    RestClient client = RestClient.create();

    String token = null;
    try {
        AuthResponse response = client.post()
            .uri(oAuthURL)
            .contentType(MediaType.APPLICATION_FORM_URLENCODED)
            .accept(MediaType.APPLICATION_JSON)
            .body(credentials)
            .retrieve()
            .body(AuthResponse.class);

        token = response.getAccessToken();
    } catch (Exception e) {
        log.error("Error in getting token {}", e.toString(), e);
    }

    return token;
}

public Map<String, List<String>> getLogFileByName(String token, 
                                                  String salesforceInstanceUrl, 
                                                  String logFileVersion) {

    Map<String, List<String>> eventMap = new HashMap<>();

    if (token == null) {
        return eventMap;
    }

    // Build the query URL
    String queryUrl = salesforceInstanceUrl + String.format(
        "/services/data/%s/query?q=SELECT+Id,+EventType,+LogDate,+LogFileLength,+LogFile+FROM+EventLogFile+ORDER+BY+CreatedDate+DESC+Nulls+Last",
        logFileVersion
    );

    // Create RestClient
    RestClient client = RestClient.create();

    try {
        EventLog response = client.get()
            .uri(queryUrl)
            .accept(MediaType.APPLICATION_JSON)
            .header(HttpHeaders.AUTHORIZATION, String.format("Bearer %s", token))
            .retrieve()
            .body(EventLog.class);

        for (EventLogRecord record : response.getRecords()) {
            eventMap.computeIfAbsent(record.getEventType(), k -> new ArrayList<>())
                    .add(record.getId());
        }

    } catch (Exception e) {
        log.error("Error in fetching log file {}", e.toString(), e);
    }

    return eventMap;
}

public String getLogFileContent(String bearerToken, 
                                String id, 
                                String salesforceInstanceUrl, 
                                String logFileVersion) {

    log.info("File id is -> {}", id);

    // Build the URL
    String url = salesforceInstanceUrl + String.format(
        "/services/data/%s/sobjects/EventLogFile/%s/LogFile", 
        logFileVersion, id
    );

    // Create RestClient
    RestClient client = RestClient.create();

    String resultOut = null;
    try {
        resultOut = client.get()
            .uri(url)
            .accept(MediaType.APPLICATION_JSON)
            .header(HttpHeaders.AUTHORIZATION, String.format("Bearer %s", bearerToken))
            .retrieve()
            .body(String.class);
    } catch (Exception e) {
        log.error("Error in fetching log file content {}", e.toString(), e);
    }

    return resultOut;
}
