import org.apache.commons.io.FileUtils;
import org.springframework.http.HttpHeaders;
import org.springframework.http.MediaType;
import org.springframework.stereotype.Component;
import org.springframework.util.StreamUtils;
import org.springframework.web.client.RestClient;

import java.io.File;
import java.io.FileOutputStream;
import java.io.IOException;

@Component
public class SalesforceLogFileDownloader {

    private final RestClient restClient;

    public SalesforceLogFileDownloader(RestClient.Builder builder) {
        this.restClient = builder.build();
    }

    public String getLogFileContentInFile(String bearerToken,
                                          String id,
                                          String salesforceInstanceUrl,
                                          String logFileVersion,
                                          String archiveFolderLocation) {

        String fileName = archiveFolderLocation + File.separator + id + ".csv";
        File destFile = new File(fileName);

        // Build the URL
        String url = salesforceInstanceUrl +
                String.format("/services/data/%s/sobjects/EventLogFile/%s/LogFile", logFileVersion, id);

        try {
            // Execute GET request
            byte[] responseBody = restClient.get()
                    .uri(url)
                    .header(HttpHeaders.AUTHORIZATION, "Bearer " + bearerToken)
                    .accept(MediaType.APPLICATION_OCTET_STREAM)
                    .retrieve()
                    .body(byte[].class);

            // Write to temp file
            File tempFile = File.createTempFile("download", ".csv");
            try (FileOutputStream fos = new FileOutputStream(tempFile)) {
                StreamUtils.copy(responseBody, fos);
            }

            // Copy to destination
            FileUtils.copyFile(tempFile, destFile);

        } catch (IOException e) {
            // Use your logger here
            System.err.println("Error in copying the file => " + fileName);
        }

        return fileName;
    }
}
